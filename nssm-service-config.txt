# NSSM Service Configuration for Your Feedback Hub
# Updated configuration to use start-service.bat
#
# PREREQUISITES:
# - Run: npm install
# - Run: npm run build (creates dist folder)
# - Download NSSM from https://nssm.cc/download
#
# INSTALLATION STEPS:
# 1. Open Command Prompt as Administrator
# 2. Navigate to your project directory
# 3. Run the commands below (adjust paths as needed):

# Install service using the batch file
nssm install "YourFeedbackHub" "C:\Users\admin\Desktop\YSM\your-feedback-hub\start-service.bat"

# Configure working directory
nssm set "YourFeedbackHub" AppDirectory "C:\Users\admin\Desktop\YSM\your-feedback-hub"

# Set service display information
nssm set "YourFeedbackHub" DisplayName "Your Feedback Hub"
nssm set "YourFeedbackHub" Description "Student Feedback Management System"

# Configure auto-start
nssm set "YourFeedbackHub" Start SERVICE_AUTO_START

# Configure logging
nssm set "YourFeedbackHub" AppStdout "C:\Users\admin\Desktop\YSM\your-feedback-hub\logs\output.log"
nssm set "YourFeedbackHub" AppStderr "C:\Users\admin\Desktop\YSM\your-feedback-hub\logs\error.log"
nssm set "YourFeedbackHub" AppRotateFiles 1
nssm set "YourFeedbackHub" AppRotateOnline 1
nssm set "YourFeedbackHub" AppRotateSeconds 86400

# Start the service
nssm start "YourFeedbackHub"

# ALTERNATIVE: Using Node.js directly (without batch file)
# nssm install "YourFeedbackHub" "C:\Program Files\nodejs\node.exe" "server.js"
# nssm set "YourFeedbackHub" AppDirectory "C:\Users\admin\Desktop\YSM\your-feedback-hub"

# SERVICE MANAGEMENT:
# View status:    nssm status YourFeedbackHub
# Stop service:   nssm stop YourFeedbackHub
# Restart:        nssm restart YourFeedbackHub
# Remove service: nssm remove YourFeedbackHub confirm
# Edit config:    nssm edit YourFeedbackHub
